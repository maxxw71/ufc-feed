   311	def historical_price_band(american_odds):
   312	    if american_odds is None or pd.isna(american_odds):
   313	        return None
   314	    o = float(american_odds)
   315	
   316	    # Historical qualifying favorites were overwhelmingly in negative-moneyline
   317	    # territory. If a future hybrid signal is -199 or better, label it as
   318	    # outside the historical buckets rather than inventing a statistic.
   319	    if o > -200:
   320	        return {
   321	            "name": "-199 or better",
   322	            "bets": 0,
   323	            "win_rate": None,
   324	            "roi": None,
   325	            "profit_50": None,

   327	        }

   329	    for band in HYBRID_PRICE_BANDS:
   330	        if band["min_odds"] <= o <= band["max_odds"]:
   331	            return dict(band)
   332	    return None

   397	def compute_hybrid_signal(p):
   398	    """
   399	    Live hybrid rule derived from the historical robustness test:
   400	      sportsbook no-vig favorite >= 70%
   401	      AND favorite is >= 3.0 years younger.
   402	    Strong tier: favorite is >= 4.0 years younger.
   403	
   404	    Historical win rate/ROI shown in email are properties of the historical
   405	    RULE, not a prediction that this individual fight has an 82.91% chance.
   406	    """
   407	    if not p.get("market_ok"):
   408	        return {"tier": "none"}
   409	
   410	    age_a = p.get("age_a")
   411	    age_b = p.get("age_b")
   412	    if age_a is None or age_b is None or pd.isna(age_a) or pd.isna(age_b):
   413	        return {"tier": "none"}
   414	
   415	    ma = float(p["market_a"])
   416	    mb = float(p["market_b"])
   417	
   418	    if ma >= mb:
   419	        favorite = str(p.get("fighter_a") or "")
   420	        underdog = str(p.get("fighter_b") or "")
   421	        favorite_side = "a"
   422	        market_prob = ma
   423	        favorite_age = float(age_a)
   424	        underdog_age = float(age_b)
   425	    else:
   426	        favorite = str(p.get("fighter_b") or "")
   427	        underdog = str(p.get("fighter_a") or "")
   428	        favorite_side = "b"
   429	        market_prob = mb
   430	        favorite_age = float(age_b)
   431	        underdog_age = float(age_a)
   432	
   433	    younger_advantage = underdog_age - favorite_age
   434	
   435	    tier = "none"
   436	    if market_prob >= 0.70 and younger_advantage >= 4.0:
   437	        tier = "strong"
   438	    elif market_prob >= 0.70 and younger_advantage >= 3.0:
   439	        tier = "qualifies"
   440	
   441	    return {
   442	        "tier": tier,
   443	        "favorite": favorite,
   444	        "underdog": underdog,
   445	        "favorite_side": favorite_side,
   446	        "market_prob": market_prob,
   447	        "favorite_age": favorite_age,
   448	        "underdog_age": underdog_age,
   449	        "younger_advantage_years": younger_advantage,
   450	    }

   639	def predict_bout(bout, event_date, bundle, reach_index=None, skill_index=None):
   640	    a = str(bout.get("fighter_a") or "").strip()
   641	    b = str(bout.get("fighter_b") or "").strip()
   642	    division = str(bout.get("division") or "Unknown").strip()
   643	    ida, idb = resolve(a, bundle), resolve(b, bundle)
   644	
   645	    if ida is None or idb is None:
   646	        return {**bout, "status": "unavailable",
   647	                "reason": "fighter not yet in historical model: " +
   648	                          ", ".join(x for x, fid in [(a, ida), (b, idb)] if fid is None)}
   649	
   650	    fab = feat_for(ida, idb, a, b, event_date, division, bundle)
   651	    fba = feat_for(idb, ida, b, a, event_date, division, bundle)
   652	    pab = model_prob(bundle["age_model"], bundle["age_calibrator"], fab, bundle["age_cols"])
   653	    pba = model_prob(bundle["age_model"], bundle["age_calibrator"], fba, bundle["age_cols"])
   654	    pa = min(max((pab + (1-pba))/2, .01), .99)
   655	
   656	    out = {
   657	        **bout, "status": "ok",
   658	        "p_a": pa, "p_b": 1-pa,
   659	        "fair_a": fair_american(pa), "fair_b": fair_american(1-pa),
   660	        "age_a": fab.get("age_a"), "age_b": fab.get("age_b"),
   661	        "age_gap": fab.get("age_gap"),
   662	    }
   663	
   664	    market = bout.get("market") or {}
   665	    if market.get("status") == "ok":
   666	        ma = float(market["consensus_no_vig_a"])
   667	        mb = float(market["consensus_no_vig_b"])
   668	        boa = float(market["best_odds_a"])
   669	        bob = float(market["best_odds_b"])
   670	        out.update({
   671	            "market_ok": True,
   672	            "market_a": ma, "market_b": mb,
   673	            "edge_a_pp": (pa-ma)*100,
   674	            "edge_b_pp": ((1-pa)-mb)*100,
   675	            "best_odds_a": boa, "best_book_a": market.get("best_book_a"),
   676	            "best_odds_b": bob, "best_book_b": market.get("best_book_b"),
   677	            "ev_a_pct": (pa * decimal_from_american(boa) - 1) * 100,
   678	            "ev_b_pct": ((1-pa) * decimal_from_american(bob) - 1) * 100,
   679	            "books": market.get("bookmakers"),
   680	        })
   681	    else:
   682	        out["market_ok"] = False
   683	
   684	    out["hybrid"] = compute_hybrid_signal(out)
   685	
   686	    reach_index = reach_index or {}
   687	    reach_a = resolve_reach(a, reach_index)
   688	    reach_b = resolve_reach(b, reach_index)
   689	    out["reach_a"] = reach_a
   690	    out["reach_b"] = reach_b
   691	
   692	    reach_hybrid = {
   693	        "tier": None,
   694	        "favorite": None,
   695	        "favorite_reach": None,
   696	        "opponent_reach": None,
   697	        "reach_advantage_in": None,
   698	        "market_prob": None,
   699	    }
   700	
   701	    market_prob = None
   702	    favorite = None
   703	    if out.get("market_ok"):
   704	        ma = float(out.get("market_a"))
   705	        mb = float(out.get("market_b"))
   706	        if ma >= mb:
   707	            market_prob = ma
   708	            favorite = a
   709	        else:
   710	            market_prob = mb
   711	            favorite = b
   712	
   713	    if market_prob is not None and favorite and reach_a is not None and reach_b is not None:
   714	        if favorite == a:
   715	            fav_reach, opp_reach = reach_a, reach_b
   716	        elif favorite == b:
   717	            fav_reach, opp_reach = reach_b, reach_a
   718	        else:
   719	            fav_reach = opp_reach = None
   720	
   721	        if fav_reach is not None and opp_reach is not None:
   722	            reach_adv = float(fav_reach) - float(opp_reach)
   723	            tier = None
   724	            if float(market_prob) >= 0.65 and reach_adv >= 6.0:
   725	                tier = "strong"
   726	            elif float(market_prob) >= 0.65 and reach_adv >= 4.0:
   727	                tier = "qualifies"
   728	
   729	            reach_hybrid.update({
   730	                "tier": tier,
   731	                "favorite": favorite,
   732	                "favorite_reach": float(fav_reach),
   733	                "opponent_reach": float(opp_reach),
   734	                "reach_advantage_in": reach_adv,
   735	                "market_prob": float(market_prob),
   736	            })
   737	    out["reach_hybrid"] = reach_hybrid
   738	
   739	    structural_edge = {
   740	        "qualifies": False,
   741	        "favorite": None,
   742	        "market_prob": None,
   743	        "younger_advantage_years": None,
   744	        "reach_advantage_in": None,
   745	    }
   746	
   747	    # Structural Edge:
   748	    # market favorite >=65%, >=2 years younger, >=6" longer reach.
   749	    if market_prob is not None and favorite and reach_a is not None and reach_b is not None:
   750	        h = out.get("hybrid") or {}
   751	        age_adv = None
   752	        # compute favorite's age advantage directly from bout ages so this signal is independent
   753	        # of whether the stricter AGE HYBRID rule qualifies.
   754	        try:
   755	            if favorite == a:
   756	                fav_age = float(out.get("age_a"))
   757	                opp_age = float(out.get("age_b"))
   758	                fav_reach, opp_reach = reach_a, reach_b
   759	            elif favorite == b:
   760	                fav_age = float(out.get("age_b"))
   761	                opp_age = float(out.get("age_a"))
   762	                fav_reach, opp_reach = reach_b, reach_a
   763	            else:
   764	                fav_age = opp_age = fav_reach = opp_reach = None
   765	
   766	            if fav_age is not None and opp_age is not None:
   767	                age_adv = opp_age - fav_age
   768	            reach_adv = float(fav_reach) - float(opp_reach) if fav_reach is not None and opp_reach is not None else None
   769	
   770	            structural_edge.update({
   771	                "favorite": favorite,
   772	                "market_prob": float(market_prob),
   773	                "younger_advantage_years": age_adv,
   774	                "reach_advantage_in": reach_adv,
   775	                "qualifies": (
   776	                    float(market_prob) >= 0.65
   777	                    and age_adv is not None and age_adv >= 2.0
   778	                    and reach_adv is not None and reach_adv >= 6.0
   779	                ),
   780	            })
   781	        except Exception:
   782	            pass
   783	
   784	    out["structural_edge"] = structural_edge
   785	
   786	    premium = {
   787	        "qualifies": False,
   788	        "favorite": None,
   789	        "favorite_reach": None,
   790	        "opponent_reach": None,
   791	        "reach_advantage_in": None,
   792	        "market_prob": None,
   793	    }
   794	    h = out.get("hybrid") or {}
   795	    if h.get("tier") in {"qualifies", "strong"} and reach_a is not None and reach_b is not None:
   796	        favorite = h.get("favorite")
   797	        if favorite == a:
   798	            fav_reach, opp_reach = reach_a, reach_b
   799	        elif favorite == b:
   800	            fav_reach, opp_reach = reach_b, reach_a
   801	        else:
   802	            fav_reach = opp_reach = None
   803	        if fav_reach is not None and opp_reach is not None:
   804	            adv = float(fav_reach) - float(opp_reach)
   805	            premium.update({
   806	                "favorite": favorite,
   807	                "favorite_reach": float(fav_reach),
   808	                "opponent_reach": float(opp_reach),
   809	                "reach_advantage_in": adv,
   810	                "market_prob": h.get("market_prob"),
   811	                "qualifies": adv >= 4.0,
   812	            })
   813	    out["age_reach_premium"] = premium
   814	    skill={"available":False,"veto":False}
   815	    if favorite:
   816	        skill=compute_skill_veto(favorite, b if favorite==a else a, skill_index or {})
   817	    out["skill_veto"]=skill
   818	    if skill.get("veto"):
   819	        h=out.get("hybrid") or {}
   820	        if h.get("tier") in {"qualifies","strong"}: h["raw_tier"]=h["tier"]; h["tier"]="none"; h["skill_vetoed"]=True
   821	        rh=out.get("reach_hybrid") or {}
   822	        if rh.get("tier") in {"qualifies","strong"}: rh["raw_tier"]=rh["tier"]; rh["tier"]=None; rh["skill_vetoed"]=True
   823	        pr=out.get("age_reach_premium") or {}
   824	        if pr.get("qualifies"): pr["raw_qualifies"]=True; pr["qualifies"]=False; pr["skill_vetoed"]=True
   825	        se=out.get("structural_edge") or {}
   826	        if se.get("qualifies"): se["raw_qualifies"]=True; se["qualifies"]=False; se["skill_vetoed"]=True
   827	    return out

   850	def market_changed(old, new):
   851	    if new is None:
   852	        return False
   853	    if old is None:
   854	        return True
   855	    if abs(new["a"] - old.get("a", new["a"])) >= .02:
   856	        return True
   857	    if abs(new["b"] - old.get("b", new["b"])) >= .02:
   858	        return True
   859	    if abs(new["oa"] - old.get("oa", new["oa"])) >= 20:
   860	        return True
   861	    if abs(new["ob"] - old.get("ob", new["ob"])) >= 20:
   862	        return True
   863	    return False
