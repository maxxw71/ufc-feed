def compute_skill_veto(favorite,opponent,idx):
    fp,op=resolve_skill(favorite,idx or {}),resolve_skill(opponent,idx or {})
    r={"available":bool(fp and op),"veto":False,"sig_diff_gap":None,"td_def_gap":None,"striking_veto":False,"wrestling_veto":False}
    if not fp or not op: return r
    if fp.get("sig_diff_pm") is not None and op.get("sig_diff_pm") is not None:
        r["sig_diff_gap"]=float(fp["sig_diff_pm"])-float(op["sig_diff_pm"])
        r["striking_veto"]=r["sig_diff_gap"]<=SKILL_VETO_SIG_DIFF
    if fp.get("td_def") is not None and op.get("td_def") is not None:
        r["td_def_gap"]=float(fp["td_def"])-float(op["td_def"])
        r["wrestling_veto"]=r["td_def_gap"]<=SKILL_VETO_TD_DEF_DIFF
    r["veto"]=bool(r["striking_veto"] or r["wrestling_veto"])
    return r

--- compute_hybrid_signal ---
def compute_hybrid_signal(p):
    """
    Live hybrid rule derived from the historical robustness test:
      sportsbook no-vig favorite >= 70%
      AND favorite is >= 3.0 years younger.
    Strong tier: favorite is >= 4.0 years younger.

    Historical win rate/ROI shown in email are properties of the historical
    RULE, not a prediction that this individual fight has an 82.91% chance.
    """
    if not p.get("market_ok"):
        return {"tier": "none"}

    age_a = p.get("age_a")
    age_b = p.get("age_b")
    if age_a is None or age_b is None or pd.isna(age_a) or pd.isna(age_b):
        return {"tier": "none"}

    ma = float(p["market_a"])
    mb = float(p["market_b"])

    if ma >= mb:
        favorite = str(p.get("fighter_a") or "")
        underdog = str(p.get("fighter_b") or "")
        favorite_side = "a"
        market_prob = ma
        favorite_age = float(age_a)
        underdog_age = float(age_b)
    else:
        favorite = str(p.get("fighter_b") or "")
        underdog = str(p.get("fighter_a") or "")
        favorite_side = "b"
        market_prob = mb
        favorite_age = float(age_b)
        underdog_age = float(age_a)

    younger_advantage = underdog_age - favorite_age

    tier = "none"
    if market_prob >= 0.70 and younger_advantage >= 4.0:
        tier = "strong"
    elif market_prob >= 0.70 and younger_advantage >= 3.0:
        tier = "qualifies"

    return {
        "tier": tier,
        "favorite": favorite,
        "underdog": underdog,
        "favorite_side": favorite_side,
        "market_prob": market_prob,
        "favorite_age": favorite_age,
        "underdog_age": underdog_age,
        "younger_advantage_years": younger_advantage,
    }

def predict_bout(bout, event_date, bundle, reach_index=None, skill_index=None):
    a = str(bout.get("fighter_a") or "").strip()
    b = str(bout.get("fighter_b") or "").strip()
    division = str(bout.get("division") or "Unknown").strip()
    ida, idb = resolve(a, bundle), resolve(b, bundle)

    if ida is None or idb is None:
        return {**bout, "status": "unavailable",
                "reason": "fighter not yet in historical model: " +
                          ", ".join(x for x, fid in [(a, ida), (b, idb)] if fid is None)}

    fab = feat_for(ida, idb, a, b, event_date, division, bundle)
    fba = feat_for(idb, ida, b, a, event_date, division, bundle)
    pab = model_prob(bundle["age_model"], bundle["age_calibrator"], fab, bundle["age_cols"])
    pba = model_prob(bundle["age_model"], bundle["age_calibrator"], fba, bundle["age_cols"])
    pa = min(max((pab + (1-pba))/2, .01), .99)

    out = {
        **bout, "status": "ok",
        "p_a": pa, "p_b": 1-pa,
        "fair_a": fair_american(pa), "fair_b": fair_american(1-pa),
        "age_a": fab.get("age_a"), "age_b": fab.get("age_b"),
        "age_gap": fab.get("age_gap"),
    }

    market = bout.get("market") or {}
    if market.get("status") == "ok":
        ma = float(market["consensus_no_vig_a"])
        mb = float(market["consensus_no_vig_b"])
        boa = float(market["best_odds_a"])
        bob = float(market["best_odds_b"])
        out.update({
            "market_ok": True,
            "market_a": ma, "market_b": mb,
            "edge_a_pp": (pa-ma)*100,
            "edge_b_pp": ((1-pa)-mb)*100,
            "best_odds_a": boa, "best_book_a": market.get("best_book_a"),
            "best_odds_b": bob, "best_book_b": market.get("best_book_b"),
            "ev_a_pct": (pa * decimal_from_american(boa) - 1) * 100,
            "ev_b_pct": ((1-pa) * decimal_from_american(bob) - 1) * 100,
            "books": market.get("bookmakers"),
        })
    else:
        out["market_ok"] = False

    out["hybrid"] = compute_hybrid_signal(out)

    reach_index = reach_index or {}
    reach_a = resolve_reach(a, reach_index)
    reach_b = resolve_reach(b, reach_index)
    out["reach_a"] = reach_a
    out["reach_b"] = reach_b

    reach_hybrid = {
        "tier": None,
        "favorite": None,
        "favorite_reach": None,
        "opponent_reach": None,
        "reach_advantage_in": None,
        "market_prob": None,
    }

    market_prob = None
    favorite = None
    if out.get("market_ok"):
        ma = float(out.get("market_a"))
        mb = float(out.get("market_b"))
        if ma >= mb:
            market_prob = ma
            favorite = a
        else:
            market_prob = mb
            favorite = b

    if market_prob is not None and favorite and reach_a is not None and reach_b is not None:
        if favorite == a:
            fav_reach, opp_reach = reach_a, reach_b
        elif favorite == b:
            fav_reach, opp_reach = reach_b, reach_a
        else:
            fav_reach = opp_reach = None

        if fav_reach is not None and opp_reach is not None:
            reach_adv = float(fav_reach) - float(opp_reach)
            tier = None
            if float(market_prob) >= 0.65 and reach_adv >= 6.0:
                tier = "strong"
            elif float(market_prob) >= 0.65 and reach_adv >= 4.0:
                tier = "qualifies"

            reach_hybrid.update({
                "tier": tier,
                "favorite": favorite,
                "favorite_reach": float(fav_reach),
                "opponent_reach": float(opp_reach),
                "reach_advantage_in": reach_adv,
                "market_prob": float(market_prob),
            })
    out["reach_hybrid"] = reach_hybrid

    structural_edge = {
        "qualifies": False,
        "favorite": None,
        "market_prob": None,
        "younger_advantage_years": None,
        "reach_advantage_in": None,
    }

    # Structural Edge:
    # market favorite >=65%, >=2 years younger, >=6" longer reach.
    if market_prob is not None and favorite and reach_a is not None and reach_b is not None:
        h = out.get("hybrid") or {}
        age_adv = None
        # compute favorite's age advantage directly from bout ages so this signal is independent
        # of whether the stricter AGE HYBRID rule qualifies.
        try:
            if favorite == a:
                fav_age = float(out.get("age_a"))
                opp_age = float(out.get("age_b"))
                fav_reach, opp_reach = reach_a, reach_b
            elif favorite == b:
                fav_age = float(out.get("age_b"))
                opp_age = float(out.get("age_a"))
                fav_reach, opp_reach = reach_b, reach_a
            else:
                fav_age = opp_age = fav_reach = opp_reach = None

            if fav_age is not None and opp_age is not None:
                age_adv = opp_age - fav_age
            reach_adv = float(fav_reach) - float(opp_reach) if fav_reach is not None and opp_reach is not None else None

            structural_edge.update({
                "favorite": favorite,
                "market_prob": float(market_prob),
                "younger_advantage_years": age_adv,
                "reach_advantage_in": reach_adv,
                "qualifies": (
                    float(market_prob) >= 0.65
                    and age_adv is not None and age_adv >= 2.0
                    and reach_adv is not None and reach_adv >= 6.0
                ),
            })
        except Exception:
            pass

    out["structural_edge"] = structural_edge

    premium = {
        "qualifies": False,
        "favorite": None,
        "favorite_reach": None,
        "opponent_reach": None,
        "reach_advantage_in": None,
        "market_prob": None,
    }
    h = out.get("hybrid") or {}
    if h.get("tier") in {"qualifies", "strong"} and reach_a is not None and reach_b is not None:
        favorite = h.get("favorite")
        if favorite == a:
            fav_reach, opp_reach = reach_a, reach_b
        elif favorite == b:
            fav_reach, opp_reach = reach_b, reach_a
        else:
            fav_reach = opp_reach = None
        if fav_reach is not None and opp_reach is not None:
            adv = float(fav_reach) - float(opp_reach)
            premium.update({
                "favorite": favorite,
                "favorite_reach": float(fav_reach),
                "opponent_reach": float(opp_reach),
                "reach_advantage_in": adv,
                "market_prob": h.get("market_prob"),
                "qualifies": adv >= 4.0,
            })
    out["age_reach_premium"] = premium
    skill={"available":False,"veto":False}
    if favorite:
        skill=compute_skill_veto(favorite, b if favorite==a else a, skill_index or {})
    out["skill_veto"]=skill
    if skill.get("veto"):
        h=out.get("hybrid") or {}
        if h.get("tier") in {"qualifies","strong"}: h["raw_tier"]=h["tier"]; h["tier"]="none"; h["skill_vetoed"]=True
        rh=out.get("reach_hybrid") or {}
        if rh.get("tier") in {"qualifies","strong"}: rh["raw_tier"]=rh["tier"]; rh["tier"]=None; rh["skill_vetoed"]=True
        pr=out.get("age_reach_premium") or {}
        if pr.get("qualifies"): pr["raw_qualifies"]=True; pr["qualifies"]=False; pr["skill_vetoed"]=True
        se=out.get("structural_edge") or {}
        if se.get("qualifies"): se["raw_qualifies"]=True; se["qualifies"]=False; se["skill_vetoed"]=True
    return out
