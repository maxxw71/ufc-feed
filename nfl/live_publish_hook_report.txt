def clean(x):
 if isinstance(x,dict):return {k:clean(v) for k,v in x.items()}
 if isinstance(x,(list,tuple)):return [clean(v) for v in x]

def publish_nfl(records,now,design):
 cards=[]
 for r in records:
  side=r.get('selection_side','home');selected=r[side];o=r['odds'];methods=[]
  for rule in r['rules']:
   mid,label,win,sample,roi,basis,definition=design.METHODS[rule]
   methods.append(dict(id=mid,title=label,win=win,sample=sample,roi=roi,note=basis,definition=definition,checks=design.evidence(r,rule),consistency=design.consistency(rule)))
  cards.append(dict(id='NFL:'+r['game_id'],start=r['kickoff'],selection=design.name(selected),opponent=design.name(r['away'] if side=='home' else r['home']),fixture=f"{design.name(r['away'])} at {design.name(r['home'])}",event=f"NFL · Week {r['week']}",price=o.get('moneyline'),book=o.get('bookmaker'),quote_at=o.get('retrieved_at'),methods=methods))

def publish_ufc(events,predictions,w,now=None):
