 'stricter':('M2','Home opener · Larger record advantage','82.4%','56 / 68','+11.1%','Archived opening prices','All M1 conditions, plus a winning-percentage advantage greater than 12.5 percentage points.'),
 'pass_defense_offense24':('M3','Week 1 · Defense + offense filter','83.7%','41 / 49','+26.5%','Archived opening prices','Week 1 home opener; previous-season pass defense top 10, sack/QB-hit rank top 16, and offense rank 24 or better.'),

Exclude only if the opponent has both top-eight sack avoidance and more rest.')}
def name(t):return TEAMS.get(t,t)
def evidence(r,rule):
 if rule in ('original','stricter'):
  h,a=r['home_record'],r['away_record']
  rows=[f"First non-neutral regular-season home game · Week {r['week']}",f"Previous-season run defense: #{r['rank']:g} of 32 (required: top 10)",f"Previous-season records: {name(r['home'])} {h['wins']}–{h['losses']}–{h['ties']} vs {name(r['away'])} {a['wins']}–{a['losses']}–{a['ties']}",f"Winning-percentage advantage: {r['record_gap_pp']:.2f} points (required: {'>12.5' if rule=='stricter' else '>0'})"]
 elif rule=='pass_defense_offense24':
  f=r['pass_defense_profile'];rows=[f"Week {r['week']} home opener",f"Previous-season pass defense: #{f['pass_rank']:g} (required: ≤10)",f"Previous-season sack/QB-hit rank: #{f['hit_rank']:g} (required: ≤16)",f"Previous-season offense: #{f['offense_rank']:g} (required: ≤24)"]
 else:
  f=r['late_profile'];rows=[f"Designated away team · Week {r['week']} (required: 11–18)",f"Pass defense: #{f['defense_rank']:g} vs #{f['opponent_defense_rank']:g} · {f['defense_edge']:g}-place advantage (required: ≥8)",f"Offensive-line continuity: {f['ol_continuity']:.0%} (required: ≥60%)",f"Prior games: {f['prior_games']} vs {f['opponent_prior_games']} (required: ≥4 each)",f"Opponent sack avoidance: #{f['opponent_protection_rank']:g} · selected team's rest advantage: {f['rest_edge']:+g} days",'Combined protection/rest exclusion: not triggered','Fresh moneyline passes −400 inclusive to below −200']
 return rows
def consistency(rule):
 if rule=='pass_defense_offense24':return '15 profitable seasons · 3 losing · 1 with no bets. 2017–2025 subset: 18/21 wins, +21.2% ROI.'
 if rule=='late_road_protection_rest_v1':return '8/8 profitable seasons (2018–2025). 2022–2025 subset: 19/22 wins, +13.5% ROI.'
 return ''
def render(records,now):
 stamp=now.astimezone(NY).strftime('%b %d, %Y · %I:%M %p ET')
 plain=[f'NFL PICKS | {len(records)} qualifying games',stamp,'']
 cards=[]
 for r in sorted(records,key=lambda r:r['kickoff']):
  selected=r['away'] if r.get('selection_side')=='away' else r['home'];opp=r['home'] if selected==r['away'] else r['away']
  title=f'{name(selected)} to win';fixture=f"{name(r['away'])} at {name(r['home'])}"
  when=datetime.fromisoformat(r['kickoff']).astimezone(NY).strftime('%a, %b %d · %I:%M %p ET')+f" · Week {r['week']}"
  o=r['odds'];p=o.get('moneyline');priced=p is not None
  badge='ODDS AVAILABLE' if priced else 'CHECK PRICE · METRICS ONLY'
  if priced:
   implied=abs(p)/(abs(p)+100) if p<0 else 100/(100+p)
   profit=10000/abs(p) if p<0 else p
   quote=f"{p:+g} · {o['bookmaker']}"
   observed=datetime.fromisoformat(o['retrieved_at']).astimezone(NY).strftime('%b %d, %I:%M %p ET')
   detail=f"Example only: $100 risk returns ${profit:.2f} profit if it wins; $100 lost if it loses. Break-even win rate at this price: {implied:.1%}."
   quote_note=f'Observed {observed}. Confirm your sportsbook’s current price.'
  else:
   quote='Current moneyline unavailable';detail='The statistical conditions match. Check your sportsbook’s price before deciding.';quote_note='No current payout calculation available.'
  plain += [title.upper(),'Moneyline · '+fixture,when,badge,quote,detail,quote_note,'']
  methods=[]
  for rule in r['rules']:
   mid,label,win,sample,roi,basis,definition=METHODS[rule];checks=evidence(r,rule);extra=consistency(rule)

   rows=''.join(f'<div style="margin:7px 0;color:#334155;">✓ {e(c)}</div>' for c in checks)

<div style="font-size:12px;font-weight:bold;letter-spacing:1px;margin-bottom:8px;">WHY THIS GAME QUALIFIES</div><div style="font-size:14px;line-height:1.6;">{rows}</div>
{('<div style="margin-top:12px;font-size:12px;color:#64748b;line-height:1.6;">'+e(extra)+'</div>') if extra else ''}</td></tr></table>''')
  overlap='<p style="font-size:12px;color:#64748b;">Multiple matching methods support the same selection. Their samples overlap; do not add their percentages or count this as multiple bets.</p>' if len(r['rules'])>1 else ''
  cards.append(f'''<table role="presentation" width="100%" cellpadding="0" cellspacing="0" style="background:#ffffff;border:1px solid #dbe3ee;border-radius:12px;margin-bottom:22px;"><tr><td style="padding:24px;">

 footer='Historical percentages and ROI describe past qualifying groups at archived prices—not this game’s predicted probability or expected return. Filters were selected retrospectively. ROI means net profit divided by total amount risked. Regular season only.'
